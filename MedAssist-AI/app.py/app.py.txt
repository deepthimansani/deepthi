# ==========================================================
# 🔑 GEMINI API
# ==========================================================

API_KEY = os.getenv("GEMINI_API_KEY")

client = genai.Client(api_key=API_KEY)

# ==========================================================
# 🧠 SYSTEM PROMPT
# ==========================================================

SYSTEM_PROMPT = """
You are MedAssist AI — a professional AI assistant
for doctors, hospitals, clinics, and medical students.

Responsibilities:
- Explain diseases
- Analyze symptoms
- Explain treatments
- Analyze medical images
- Summarize reports
- Provide medical education

Rules:
- Be professional
- Use headings and bullet points
- Never replace licensed doctors
- Mention limitations when needed

Always end with:
⚠️ This AI is for educational and support purposes only.
"""

# ==========================================================
# 🤖 AI FUNCTION
# ==========================================================

def medical_ai(question, image, department):

    try:

        contents = []

        # Upload image if available
        if image:
            uploaded_file = client.files.upload(file=image)
            contents.append(uploaded_file)

        contents.append(f"""
Department: {department}

Question:
{question}
""")

        response = client.models.generate_content(
            model="gemini-2.5-flash",
            contents=contents,
            config={
                "system_instruction": SYSTEM_PROMPT,
                "temperature": 0.4
            }
        )

        time_now = datetime.now().strftime("%I:%M %p")

        return f"""
# 🩺 MedAssist AI Response

⏰ {time_now}

{response.text}
"""

    except Exception as e:
        return f"❌ Error: {str(e)}"

# ==========================================================
# 💬 CHAT FUNCTION
# ==========================================================

def chat_function(message, history):

    response = medical_ai(
        message,
        None,
        "General Medicine"
    )

    history.append((message, response))

    return "", history
